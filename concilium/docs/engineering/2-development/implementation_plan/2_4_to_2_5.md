**Issue: Create concilium/optimizer/loop.py**

- [ ] **Define class structure**:

  ```python
  class PlanningLoop:
      """Orchestrator above skills. Composes DV1–DV4 against hard constraints."""
      
      def __init__(self, runtime: Runtime):
          self.runtime = runtime
          
      def adjust_schedule(
          self,
          canonical: CanonicalModel,
          target_decisions: DecisionVariables,
      ) -> tuple[CanonicalModel, bool, list[ConstraintViolation]]:
          """
          Adjust schedule + resources + costs to satisfy hard constraints (C1–C5).
          
          Returns:
              (updated_canonical, feasible, violations)
              
          If feasible=True, all C1–C5 pass; canonical is usable.
          If feasible=False, violations list shows which hard constraints failed.
          """
  ```

- [ ] **Implement core loop (L1 — Scheduling + Resources)**:
  ```python
  def adjust_schedule(self, canonical, target):
      # 1. Apply DV3 (start_offsets) to canonical.activity_dag
      #    → verify precedence still holds
      
      # 2. Apply DV2 (num_equipes) to skill 02 inputs
      #    → run cronograma skill
      #    → check C2 (prazo) and C3 (resources)
      
      # 3. If C2 or C3 fail → return feasible=False with violations
      
      # 4. Otherwise → canonical is updated; continue to cost loops
      
      violations = []
      
      # Apply offsets
      canonical.decision_variables.dv3_start_offset = target.dv3_start_offset
      dag_violations = self._validate_dag_consistency(canonical)
      violations.extend(dag_violations)
      if any(v.status == "ERRO" for v in violations):
          return (canonical, False, violations)
      
      # Run cronograma with new offsets
      skill_result = self.runtime.run_skill(
          "02-cronograma",
          canonical,
          {
              "num_equipes_por_item": target.dv2_num_equipes,
              "start_offsets": target.dv3_start_offset,
          }
      )
      canonical = skill_result.canonical
      violations.extend(skill_result.violations)
      
      # Check hard constraints C1–C5
      hard_violations = [v for v in violations if v.status == "ERRO"]
      if hard_violations:
          return (canonical, False, violations)
      
      return (canonical, True, violations)
  ```

- [ ] **Implement L2 (Cost equilibrium)**:

  ```python
  def _balance_costs(self, canonical, target):

              magnitude=delta,
              remediation_hint="Adjust crew template or productivity factors"
          )
      return None
  ```

- [ ] **Implement L4 (Economic evaluation)**:

  ```python
  def _evaluate_economics(self, canonical):

          "06-consolidar",
          canonical,
          {}
      )
      return (skill_result.canonical, skill_result.violations)
  ```

- [ ] **Main entry point — compose all loops**:
  ```python
  def adjust_schedule(self, canonical, target_decisions):
      violations = []
      
      # L1: Scheduling + Resources
      canonical, feasible, loop1_viol = self._schedule_with_resources(
          canonical, target_decisions
      )
      violations.extend(loop1_viol)
      if not feasible:
          return (canonical, False, violations)
      
      # L2: Cost equilibrium
      balance_viol = self._balance_costs(canonical, target_decisions)
      if balance_viol:
          violations.append(balance_viol)
      
      # L3: Indirect policy
      canonical, loop3_viol = self._apply_indirect_policy(
          canonical, target_decisions
      )
      violations.extend(loop3_viol)
      
      # L4: Economic evaluation
      canonical, loop4_viol = self._evaluate_economics(canonical)
      violations.extend(loop4_viol)
      
      # Final check: any remaining ERRO?
      hard_fails = [v for v in violations if v.status == "ERRO"]
      feasible = len(hard_fails) == 0
      
      return (canonical, feasible, violations)
  ```
